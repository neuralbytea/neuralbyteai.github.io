# Rate card → billing amount, cost and margin per line
class AnalyticLine(models.Model):
    _inherit = 'account.analytic.line'

    def _compute_margin(self):
        for line in self:
            rate = line._find_rate_card()
            line.billing_amount = line.unit_amount * rate.billing
            line.cost_amount = line.unit_amount * rate.cost
            line.margin = line.billing_amount - line.cost_amount
